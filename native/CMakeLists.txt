cmake_minimum_required(VERSION 3.20)
project(LumaWorkbench VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenGL REQUIRED)
find_package(glfw3 3.3 QUIET)
if(NOT TARGET glfw)
 include(FetchContent)
 set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
 set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
 set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
 FetchContent_Declare(glfw URL https://github.com/glfw/glfw/archive/refs/tags/3.4.tar.gz)
 FetchContent_MakeAvailable(glfw)
endif()
add_library(luma_glad STATIC third_party/glad/src/gl.c)
target_include_directories(luma_glad PUBLIC third_party/glad/include)
add_library(luma_math INTERFACE)
target_include_directories(luma_math INTERFACE third_party/glm include)
add_library(luma_cpu STATIC src/scene/Transform.cpp src/scene/Camera.cpp src/scene/CameraControls.cpp src/scene/Scene.cpp src/scene/MaterialDesc.cpp src/gfx/MeshData.cpp src/gfx/TextureOptions.cpp src/scene/MaterialLibrary.cpp src/editor/EditorController.cpp src/lighting/LightTypes.cpp src/lighting/LightRig.cpp src/render/PostProcessSettings.cpp)
target_link_libraries(luma_cpu PUBLIC luma_math)
target_sources(luma_cpu PRIVATE src/studio/TimelineModel.cpp src/studio/PipelineState.cpp)
add_library(luma_gfx STATIC src/gfx/ShaderProgram.cpp src/gfx/Mesh.cpp src/gfx/Texture2D.cpp src/gfx/RenderState.cpp src/gfx/NativeResourceFactory.cpp)
target_include_directories(luma_gfx PRIVATE third_party/stb)
target_link_libraries(luma_gfx PUBLIC luma_cpu luma_glad OpenGL::GL)
add_library(luma_core STATIC src/core/GlfwWindow.cpp src/core/Input.cpp)
target_link_libraries(luma_core PUBLIC luma_gfx glfw)
add_library(luma_scene STATIC src/scene/AssetStore.cpp)
target_link_libraries(luma_scene PUBLIC luma_gfx)
add_library(luma_render STATIC src/render/Renderer.cpp src/render/DisplayTransfer.cpp src/render/LightingPipeline.cpp src/render/PostProcessTarget.cpp src/render/PostProcessPipeline.cpp src/render/RenderTarget.cpp)
target_link_libraries(luma_render PUBLIC luma_scene)
add_library(luma_imgui STATIC third_party/imgui/imgui.cpp third_party/imgui/imgui_draw.cpp third_party/imgui/imgui_tables.cpp third_party/imgui/imgui_widgets.cpp third_party/imgui/backends/imgui_impl_glfw.cpp third_party/imgui/backends/imgui_impl_opengl3.cpp)
target_include_directories(luma_imgui PUBLIC third_party/imgui)
target_link_libraries(luma_imgui PUBLIC glfw OpenGL::GL)
add_library(luma_presets STATIC scenes/Presets.cpp src/app/Assets.cpp)
target_link_libraries(luma_presets PUBLIC luma_render)
target_compile_definitions(luma_presets PRIVATE LUMA_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
add_library(luma_app STATIC src/app/Application.cpp src/editor/EditorUI.cpp)
target_link_libraries(luma_app PUBLIC luma_core luma_render luma_presets PRIVATE luma_imgui)
target_compile_definitions(luma_app PRIVATE LUMA_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
add_executable(luma_workbench src/app/main.cpp)
target_link_libraries(luma_workbench PRIVATE luma_app)
add_executable(luma_reuse_example examples/reusable_scene.cpp)
target_link_libraries(luma_reuse_example PRIVATE luma_app)
add_executable(luma_controller_example examples/controller_example.cpp)
target_link_libraries(luma_controller_example PRIVATE luma_cpu)
add_executable(luma_lighting_example examples/lighting_example.cpp)
target_link_libraries(luma_lighting_example PRIVATE luma_render)
include(CTest)
